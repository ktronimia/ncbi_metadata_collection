
#unwanted character containing sequence removal

seqkit grep -s -r -v -p '[^ATGCatgc]' combined.fasta > clean.fasta

# fasta to table (tsv)

awk '/^>/ {if (seq) print header "\t" seq "\t" length(seq); header=substr($0,2); seq=""; next} {seq=seq$0} END {if (seq) print header "\t" seq "\t" length(seq)}' input.fasta > table.tsv

# table(tsv) to fasta

awk -F'\t' '{print ">"$1 "\n" $2}' table.tsv > output.fasta
